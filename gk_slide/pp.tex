\section{Cơ sở lý thuyết ALPR}

\begin{frame}{Tổng quan hệ thống ALPR}
    \textbf{ALPR gồm hai giai đoạn chính:}
    \begin{enumerate}
        \item \textbf{Plate Detection:} Xác định vị trí biển số trong ảnh.
        \item \textbf{OCR:} Nhận dạng ký tự trên biển số.
    \end{enumerate}

    \begin{block}{Pipeline tổng quan}
    \centering
    \textbf{Ảnh đầu vào} $\rightarrow$ \textbf{Plate Detection} $\rightarrow$ \textbf{Tiền xử lý (CLAHE)} $\rightarrow$ \textbf{OCR} $\rightarrow$ \textbf{Biển số}
    \end{block}
\end{frame}

\begin{frame}{Định dạng biển số Việt Nam}
    \begin{table}
        \centering
        \begin{tabular}{lll}
            \toprule
            \textbf{Loại} & \textbf{Định dạng} & \textbf{Ví dụ} \\
            \midrule
            Ô tô (1 dòng) & XX-Y XXXXX & 30G-07784 \\
            Ô tô (2 dòng) & XX-Y / XXXXX & 30G / 07784 \\
            Xe máy & XX-YY / XXX.XX & 29-H1 / 234.56 \\

    \begin{itemize}
        \item 2 số đầu: mã tỉnh/thành; chữ cái: seri; 4--5 số: thứ tự đăng ký.
        \item Thách thức: đa dạng định dạng, góc quay nghiêng, ánh sáng thay đổi.
    \end{itemize}
\end{frame}

\begin{frame}{PaddleOCR và kiến trúc CRNN}
    \begin{itemize}
        \item \textbf{CNN Backbone:} Trích xuất đặc trưng không gian từ ảnh biển số.
        \item \textbf{Bi-LSTM:} Mô hình hóa quan hệ tuần tự ký tự.
        \item \textbf{CTC Decoder:} Giải mã chuỗi ký tự không cần alignment thủ công.
    \end{itemize}

    \begin{block}{CTC Loss (tóm tắt)}
    \[ P(\mathbf{l} | \mathbf{x}) = \sum_{\pi \in \mathcal{B}^{-1}(\mathbf{l})} P(\pi | \mathbf{x}) \]
    \[ \mathcal{L}_{CTC} = -\log P(\mathbf{l} | \mathbf{x}) \]
    \end{block}
\end{frame}

\begin{frame}{Tiền xử lý ảnh biển số}
    \begin{itemize}
        \item \textbf{CLAHE:} Cân bằng tương phản cục bộ, tăng độ rõ ký tự.
        \item \textbf{Sharpening:} Làm sắc nét biên ký tự trước OCR.
    \end{itemize}

    \begin{block}{Quy trình chuẩn}
    Crop biển số $\rightarrow$ Grayscale $\rightarrow$ CLAHE $\rightarrow$ RGB $\rightarrow$ Sharpening (tuỳ chọn)
    \end{block}
\end{frame}

\begin{frame}{Validation \& Retry}
    \textbf{Ý tưởng:} Chỉ lưu kết quả OCR đúng format biển số Việt Nam.
    \begin{itemize}
        \item Xe máy: \texttt{[0-9]\{2\}[A-Z][A-Z0-9][0-9]\{4,5\}}
        \item Ô tô: \texttt{[0-9]\{2\}[A-Z][0-9]\{4,5\}}
        \item Nếu sai format $\rightarrow$ retry ở frame tiếp theo.
    \end{itemize}
\end{frame}
