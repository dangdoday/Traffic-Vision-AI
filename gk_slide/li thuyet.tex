        \item ITS và camera phạt nguội bắt đầu triển khai nhưng còn phân tán.
        \item Giao thông hỗn hợp, mật độ xe máy cao, vi phạm phổ biến.

    \begin{alertblock}{Thách thức chính}

            \item Điều kiện ánh sáng thay đổi, che khuất, góc quay nghiêng.
            \item Nhu cầu làm chủ công nghệ phù hợp đặc thù Việt Nam.

\section{Cơ sở lý thuyết: Phát hiện \& Theo dõi}

\begin{frame}{Tổng quan về Phát hiện đối tượng}
    \begin{itemize}
        \item \textbf{Định nghĩa:} Xác định vị trí và phân loại các thực thể quan tâm trong ảnh/video.
        \item \textbf{Đầu ra:} Tập bounding box $B = \{b_1, b_2, \dots, b_N\}$:
        $$ b_i = (x_i, y_i, w_i, h_i, c_i, p_i) $$
        \item $(x_i, y_i)$: tọa độ tâm; $(w_i, h_i)$: kích thước; $c_i$: lớp; $p_i$: độ tin cậy.
        \item \textbf{Thách thức:} biến thiên tỉ lệ, che khuất, ánh sáng thay đổi.
    \end{itemize}
\end{frame}

\begin{frame}{Lựa chọn mô hình: YOLOv8}
    \textbf{Lý do lựa chọn YOLOv8 cho bài toán giao thông:}
    \begin{itemize}
        \item \textbf{Anchor-free:} Linh hoạt với đối tượng có tỉ lệ biến thiên lớn.
        \item \textbf{Module C2f:} Trích xuất đặc trưng giàu thông tin, nhẹ.
        \item \textbf{Task Aligned Assigner:} Cân bằng giữa phân loại và định vị.
        \item \textbf{Triển khai:} Tối ưu cho suy luận thời gian thực.
    \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Kiến trúc mạng YOLOv8}
    Hệ thống gồm 3 thành phần chính:
    \begin{enumerate}
        \item \textbf{Backbone (CSPDarknet cải tiến):} Trích xuất đặc trưng với C2f, SPPF.
        \item \textbf{Neck (PANet):} Kết hợp đặc trưng đa tỉ lệ.
        \item \textbf{Head (Decoupled):} Tách nhánh phân loại và hồi quy hộp.
    \end{enumerate}
    \framebreak
    \begin{figure}
        \centering
        \includegraphics[width=0.65\linewidth]{Figures/1698478493766.jpg}
    \end{figure}
\end{frame}

\begin{frame}{Hàm mất mát trong YOLOv8}
    Tối ưu hóa hàm mất mát tổng hợp:
    $$ L_{total} = \lambda_{box} L_{box} + \lambda_{cls} L_{cls} + \lambda_{dfl} L_{dfl} $$
    \begin{block}{Thành phần chính}
    \begin{itemize}
        \item \textbf{CIoU Loss} cho định vị khung.
        \item \textbf{DFL} cho chất lượng biên khung.
        \item \textbf{VFL/BCE} cho phân loại.
    \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Theo dõi đa đối tượng (MOT)}
    \textbf{Mục tiêu:} Gán ID duy nhất cho mỗi đối tượng qua các khung hình.
    \begin{itemize}
        \item \textbf{Kalman Filter:} Dự đoán vị trí/động học.
        \item \textbf{Hungarian Algorithm:} Ghép cặp detection và track tối ưu.
        \item \textbf{Tracking-by-Detection:} Dựa trên đầu ra YOLOv8.
    \end{itemize}
\end{frame}

\begin{frame}[allowframebreaks]{Bộ lọc Kalman trong Tracking}
    \begin{block}{Mô hình trạng thái (constant velocity)}
    $\mathbf{x}_k = [u, v, a, h, \dot{u}, \dot{v}, \dot{a}, \dot{h}]^T$
    \end{block}
    \begin{block}{Ký hiệu trong công thức}
    \begin{itemize}
        \item $\mathbf{x}_k$: trạng thái tại khung $k$; $(u,v)$ là tâm, $a=w/h$, $h$ là chiều cao.
        \item $\dot{u}, \dot{v}, \dot{a}, \dot{h}$: vận tốc của các đại lượng tương ứng.
        \item $\mathbf{F}$: ma trận chuyển trạng thái; $\mathbf{H}$: ma trận quan sát.
        \item $\mathbf{z}_k$: đo lường từ detector; $\mathbf{K}$: Kalman gain.
        \item $\hat{\mathbf{x}}_{k|k-1}$: ước lượng trước cập nhật; $\hat{\mathbf{x}}_{k|k}$: sau cập nhật.
    \end{itemize}
    \end{block}
    \begin{block}{Hai bước chính}
    \begin{itemize}
        \item \textbf{Dự đoán:} $\hat{\mathbf{x}}_{k|k-1} = \mathbf{F}\hat{\mathbf{x}}_{k-1|k-1}$
        \item \textbf{Cập nhật:} $\hat{\mathbf{x}}_{k|k} = \hat{\mathbf{x}}_{k|k-1} + \mathbf{K}(\mathbf{z}_k-\mathbf{H}\hat{\mathbf{x}}_{k|k-1})$
    \end{itemize}
    \end{block}
    \textbf{Ý nghĩa:} làm mượt quỹ đạo, dự đoán vị trí khi bị che khuất ngắn.
\end{frame}

\begin{frame}[allowframebreaks]{Thuật toán Hungarian (Kuhn--Munkres)}
    \textbf{Bài toán:} ghép 1--1 giữa $M$ track và $N$ detection để tối thiểu hóa chi phí.
    \begin{itemize}
        \item \textbf{Ma trận chi phí:} $C_{i,j} = 1 - IoU(\text{track}_i, \text{det}_j)$.
        \item \textbf{Nếu $M \neq N$:} bổ sung hàng/cột giả để thành ma trận vuông.
    \end{itemize}
    \begin{block}{Cách hoạt động}
    \begin{enumerate}
        \item \textbf{Giảm hàng:} trừ phần tử nhỏ nhất mỗi hàng.
        \item \textbf{Giảm cột:} trừ phần tử nhỏ nhất mỗi cột.
        \item \textbf{Phủ số 0:} dùng số đường thẳng tối thiểu để phủ hết các số 0.
        \item Nếu chưa phủ đủ $n$ đường, điều chỉnh ma trận (trừ/cộng phần tử nhỏ nhất chưa phủ) và lặp.
        \item \textbf{Chọn 0 độc lập:} mỗi hàng/cột chọn một số 0 $\rightarrow$ phép gán tối ưu.
    \end{enumerate}
    \end{block}
    \textbf{Độ phức tạp:} $O(n^3)$, phù hợp thời gian thực.
\end{frame}

\begin{frame}[allowframebreaks]{Thuật toán ByteTrack}
    \textbf{Ý tưởng:} Tận dụng mọi detection, kể cả điểm tin cậy thấp.
    \begin{enumerate}
        \item \textbf{Giai đoạn 1 (High Score Matching):}
        Ghép detection chất lượng cao với track hiện có (IoU + Kalman).
        \item \textbf{Giai đoạn 2 (Low Score Matching):}
        Ghép các track chưa khớp với detection điểm thấp để “cứu” ID.
    \end{enumerate}
    \textit{Ưu điểm:} Tận dụng detection điểm thấp, giảm mất ID khi che khuất.
    \framebreak
    \begin{figure}
        \centering
        \includegraphics[width=0.75\linewidth]{Figures/bytetrack.png}
        \caption{Minh họa cơ chế liên kết hai giai đoạn của ByteTrack}
    \end{figure}
\end{frame}

\begin{frame}[allowframebreaks]{Xử lý ảnh \& Hình học tính toán}
    \begin{block}{Nhận diện đèn (HSV)}
        Chuyển RGB $\to$ HSV để tách màu và độ sáng.
        \begin{itemize}
            \item \textbf{Đỏ:} $H \in [0,10] \cup [160,180]$
            \item \textbf{Vàng:} $H \in [20,35]$
            \item \textbf{Xanh:} $H \in [40,85]$
            \item Voting theo tỉ lệ pixel để quyết định trạng thái đèn.
        \end{itemize}
    \end{block}
    \framebreak
    \begin{block}{Hình học tính toán}
        \begin{itemize}
            \item \textbf{Point-in-Polygon:} Xác định xe thuộc làn/ROI.
            \item \textbf{Stopline:} Tính khoảng cách tới vạch dừng để phát hiện cắt qua.
            \item \textbf{Hướng di chuyển:} So sánh vector chuyển động với vector tham chiếu để phân loại đi thẳng/rẽ.
        \end{itemize}
    \end{block}
\end{frame}

\begin{frame}{Hình học tính toán trong phát hiện vi phạm}
    \begin{block}{Các phép kiểm tra chính}
    \begin{itemize}
        \item \textbf{Vượt đèn đỏ:} xe cắt \emph{stopline} khi tín hiệu \textbf{đỏ}.
        \item \textbf{Đi sai làn:} vị trí tâm/điểm đáy xe nằm ngoài đa giác làn hợp lệ.
        \item \textbf{Sai hướng:} góc giữa vector chuyển động và vector làn vượt ngưỡng.
    \end{itemize}
    \end{block}
    \begin{block}{Đầu vào hình học}
        ROI, đa giác làn, vạch dừng, hướng chuẩn được cấu hình theo camera.
    \end{block}
\end{frame}

\section{Cơ sở lý thuyết: Nhận diện biển số xe}
\begin{frame}{Tiền xử lý ảnh biển số}
    \textbf{Mục đích:} Cải thiện chất lượng ảnh trước khi OCR
    
    \begin{block}{Các bước tiền xử lý}
    \begin{enumerate}
        \item \textbf{Resize:} Đưa về chiều cao chuẩn (100px)
        \item \textbf{Grayscale:} Chuyển ảnh xám
        \item \textbf{CLAHE:} Cân bằng histogram thích ứng
        \item \textbf{Bilateral Filter:} Khử nhiễu giữ cạnh
        \item \textbf{Sharpening:} Làm sắc nét ký tự
    \end{enumerate}
    \end{block}
    
    \textbf{CLAHE} (Contrast Limited Adaptive Histogram Equalization):
    \begin{itemize}
        \item Chia ảnh thành tile $8 \times 8$
        \item Equalization cục bộ + clip limit để tránh noise
    \end{itemize}
\end{frame}

\begin{frame}{Tổng quan về ALPR}
    \begin{columns}[c]
    \begin{column}{0.55\textwidth}
        \textbf{ALPR} (Automatic License Plate Recognition) gồm 2 giai đoạn:
        \begin{enumerate}
            \item \textbf{Plate Detection:} Xác định vị trí biển số trong ảnh
            \item \textbf{Character Recognition (OCR):} Nhận dạng các ký tự
        \end{enumerate}
        
        \vspace{0.3em}
        \textbf{Thách thức với biển số Việt Nam:}
        \begin{itemize}
            \item Đa dạng định dạng (1 dòng, 2 dòng)
            \item Biển số bị nghiêng, mờ, bẩn
            \item Điều kiện ánh sáng thay đổi
        \end{itemize}
    \end{column}
    \begin{column}{0.42\textwidth}
        \centering
        \includegraphics[width=\linewidth]{Figures/plate_inference1.png}
        
        \tiny Minh họa vùng biển số
    \end{column}
    \end{columns}
\end{frame}

\begin{frame}{PaddleOCR - Giới thiệu}
    \textbf{PaddleOCR 2.7.x (PP-OCRv4)} - Công cụ OCR mã nguồn mở của Baidu
    
    \begin{block}{Đặc điểm chính}
    \begin{itemize}
        \item Hỗ trợ \textbf{80+ ngôn ngữ}
        \item Kiến trúc nhẹ: \textbf{SVTR\_LCNet} (~12MB model)
    \end{itemize}
    \end{block}
    
    \vspace{0.5em}

        \item \texttt{lang='en'}: Bộ ký tự Latin (A-Z, 0-9)

\begin{frame}{Pipeline nhận dạng trong PaddleOCR}
    \begin{block}{Luồng xử lý}
    \centering
    \small
    \textbf{Ảnh biển số} $\rightarrow$ \textbf{Rectification} $\rightarrow$ \textbf{Feature Extraction} $\rightarrow$ \textbf{Sequence Modeling} $\rightarrow$ \textbf{CTC Decoder}
    \end{block}
    
    \vspace{0.5em}
    \textbf{Các bước chính:}
    \begin{enumerate}
        \item \textbf{Rectification:} Nắn thẳng biển số nghiêng/cong (STN/TPS)
        \item \textbf{Feature Extraction:} Trích xuất đặc trưng (SVTR backbone)
        \item \textbf{Sequence Modeling:} Mô hình hóa chuỗi ký tự (Attention)
        \item \textbf{CTC Decoder:} Giải mã thành chuỗi ký tự cuối cùng
    \end{enumerate}
\end{frame}

\begin{frame}{Rectification - Nắn thẳng biển số}
    \textbf{Vấn đề:} Biển số thường bị nghiêng/xoay do góc camera
    
    \begin{columns}[c]
    \begin{column}{0.48\textwidth}
        \begin{block}{Spatial Transformer Network}
        \begin{enumerate}
            \item \textbf{Localization:} CNN học $\theta$
            \item \textbf{Grid Generator:} Lưới mới
            \item \textbf{Sampler:} Lấy mẫu pixel
        \end{enumerate}
        \end{block}
        
        \textbf{Phép biến đổi Affine:}
        \small
        $$ \begin{pmatrix} x^s \\ y^s \end{pmatrix} = \begin{bmatrix} \theta_{11} & \theta_{12} & \theta_{13} \\ \theta_{21} & \theta_{22} & \theta_{23} \end{bmatrix} \begin{pmatrix} x^t \\ y^t \\ 1 \end{pmatrix} $$
    \end{column}

\end{frame}

\begin{frame}{TPS - Thin-Plate Spline}
    \textbf{Vấn đề:} Affine không đủ cho văn bản cong (curved text)
    
    \begin{block}{Thin-Plate Spline Transformation}
    Sử dụng $K$ điểm điều khiển để biến đổi phi tuyến:
    $$ f(x, y) = a_0 + a_x x + a_y y + \sum_{k=1}^{K} w_k \cdot U(r_k) $$
    \begin{itemize}
        \item $U(r) = r^2 \log(r)$: Radial basis function
        \item $K = 20$ điểm điều khiển (thường dùng)
    \end{itemize}
    \end{block}
    
    \vspace{0.3em}
    \textbf{Tác dụng:} Văn bản cong $\rightarrow$ Văn bản thẳng
\end{frame}

\begin{frame}{CTC Decoder - Giải mã chuỗi ký tự}
    \textbf{CTC} (Connectionist Temporal Classification): Giải mã không cần alignment
    
    \begin{block}{Nguyên lý}
    \begin{itemize}
        \item Mỗi ``lát cắt'' dọc $\rightarrow$ vector xác suất trên tập ký tự
        \item Thêm ký tự đặc biệt \textbf{blank} ($\epsilon$)
        \item \textbf{Collapse rule:} Loại bỏ lặp + blank
    \end{itemize}
    \end{block}
    
    \textbf{Ví dụ:} Biển số ``15A''
    \begin{center}
    \small
    \texttt{1-1-$\epsilon$-5-$\epsilon$-A-A} $\xrightarrow{\text{collapse}}$ \texttt{15A}
    \end{center}
    
    \vspace{0.3em}
    \textbf{Decoding:}
    \begin{itemize}
        \item \textbf{Greedy:} Chọn ký tự xác suất cao nhất mỗi bước
        \item \textbf{Beam Search:} Duy trì $k$ hypotheses tốt nhất
    \end{itemize}
\end{frame}
